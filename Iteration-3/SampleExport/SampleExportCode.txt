@startuml SequenceDiagram_iCalExport
title iCal Export — Filtered Task List

skinparam sequenceArrowThickness 1
skinparam sequenceParticipantBorderColor Black
skinparam sequenceParticipantBackgroundColor White
skinparam sequenceLifeLineBorderColor Black
skinparam noteBackgroundColor LightYellow
skinparam noteBorderColor Gray

actor       User               as user
participant "ExportImportMenu" as menu   <<presentation>>
participant "SearchService"    as search <<domain.service>>
participant "TaskRepository"   as repo   <<persistence>>
participant "ICalGateway"      as gw     <<domain.gateway>>
participant "ICalGatewayImpl"  as gwImpl <<gateway>>

== Specify Export Criteria ==

user  -> menu    : requestICalExport()
menu  -> user    : promptSearchCriteria()
user  -> menu    : criteria, outputFilePath

== Retrieve Matching Tasks ==

menu  -> search  : search(criteria)
search -> repo   : findByCriteria(criteria)
repo  --> search : List<Task>
search --> menu  : List<Task>

== Export to iCal ==

menu -> gwImpl : exportTasks(tasks, outputStream)

loop for each task in tasks

    alt task.dueDate != null

        gwImpl -> gwImpl : createVEVENT()
        note right
          SUMMARY   = task.title
          DTSTART   = task.dueDate
          DTEND     = task.dueDate
          DESCRIPTION = task.description
                      + priority, status
                      + projectName (if any)
                      + subtask summary (if any)
          STATUS    = mapped from TaskStatus
        end note

    else task.dueDate == null

        gwImpl -> gwImpl : skip task
        note right: Tasks without a due date\nare ignored (spec requirement)

    end

end

gwImpl -> gwImpl : writeCalendar(outputStream)
gwImpl --> menu  : void

== Confirm ==

menu -> user : "Export complete: <n> tasks written to <file>"

@enduml
