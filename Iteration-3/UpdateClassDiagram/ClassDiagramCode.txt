  OPEN
  COMPLETED
  CANCELLED
}

  LOW
  MEDIUM
  HIGH
}

  JUNIOR=10
  INTERMEDIATE=5
  SENIOR=2
  
  + setLimit(newLimit: int) : void
}

class Task {
  - id : int
  - title : String
  - description : String[0..1]
  - creationDate : Date
  - dueDate : Date[0..1]
  - priorityLevel : PriorityLevel
  - status : TaskStatus

  + complete() : void
  + cancel() : void
  + isOpen() : Boolean
  + updateTitle(title:String) : void
  + updateDescription(desc:String) : void
  + updatePriority(priority:PriorityLevel) : void
  + updateDueDate(date:Date) : void
  + createSubtask(title:String) : Subtask
  + getSubtasksSummary() : String
  + assignCollaborator(collab:Collaborator) : Subtask
}

TaskCatalogue "1" *-- "0..*" Task 

note left of Task
OCL 1:
context Task
inv MaxSubtasks:
  self.subtasks->size() <= 20
end note

note right of Task
OCL 2:
context Task
inv OpenTasksWithoutDueDateLimit:
  Task.allInstances()
    ->select(t | t.status = TaskStatus::OPEN and t.dueDate.oclIsUndefined())
    ->size() <= 50
end note

note bottom of Task
Constraint:
(title, dueDate) must be unique
end note

class Collaborator {
  - name : String

  + getAssignedSubtasks() : List<Subtask>
  + getOpenTasksCount() : int
  + getLimit() : int
  + isOverloaded() : Boolean
}

CollaboratorCatalogue "1" *-- "0..*" Collaborator

note right of Collaborator
OCL 4:
context Collaborator
inv NotOverloaded:
  self.getOpenTasksCount() <= self.getLimit()
end note

note bottom of Collaborator
Limits (global/static per category,
modifiable at runtime -> can overload instances): 
JUNIOR = 10
INTERMEDIATE = 5
SENIOR = 2
end note
