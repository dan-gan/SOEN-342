' =========================
' ENUMS
' =========================

enum TaskStatus {
  OPEN
  COMPLETED
  CANCELLED
}

enum PriorityLevel {
  LOW
  MEDIUM
  HIGH
}

enum CollaboratorStatus {
  OK
  OVERLOADED
}

' =========================
' ENTITY CLASSES
' =========================

class Task {
    -id: int
    -title: String
    -description: String[0..1]
    -creationDate: Date
    -dueDate: Date[0..1]
    -priorityLevel: PriorityLevel
    -status: TaskStatus

  + complete() : void
  + cancel() : void
  + updateTitle(title:String) : void
  + updateDescription(desc:String) : void
  + updatePriority(priority:PriorityLevel) : void
  + updateDueDate(date:LocalDateTime) : void
  + createSubtask(title:String) : Subtask
}

class Project {
  - id : int
  - name : String
  - description : String[0..1]
  
  + addTask(task:Task) : void
  + removeTask(task:Task) : void
}

class Subtask {
  - id : int
  - title : String
  - status : TaskStatus

  + complete() : void
}

class Tag {
  - id : int
  - name : String
}

class ActivityEntry {
  - timestamp : Date
  - description : String
}

' =========================
' COLLABORATORS
' =========================

abstract class Collaborator {
  - name : String
  - taskLimit : int
  - status : CollaboratorStatus

  + assignTask(task:Task) : void
  + changeTaskLimit(newLimit:int) : void
}

class JuniorCollaborator extends Collaborator
class IntermediateCollaborator extends Collaborator
class SeniorCollaborator extends Collaborator

' =========================
' RECURRENCE
' =========================

interface Recurrence {
  - startDate : Date
  - endDate : Date
}

class DailyRecurrence implements Recurrence
class WeeklyRecurrence implements Recurrence
class MonthlyRecurrence implements Recurrence

' =========================
' SYSTEM / SERVICE CLASSES
' =========================

class TaskManager {

  + createTask(title:String) : Task
  + updateTask(task:Task) : void
  + completeTask(task:Task) : void
  + cancelTask(task:Task) : void
  + searchTasks(keyword:String) : List<Task>
  + createProject(name:String) : void
}

class CSVImporter {
  + importTasks(filePath:String) : List<Task>
}

class CSVExporter {
  + exportTasks(tasks:List<Task>, filePath:String) : void
}

' =========================
' ASSOCIATIONS
' =========================

Task "0..*" -- "tasks {ordered, List}\n0..1" Project 
Task "1" *-- "subtasks\n{ordered, List}\n0..*" Subtask 
Task "1" *-- "activityEntries\n{ordered, List}\n0..*" ActivityEntry 
Task "0..*" -- "tags\n{ordered, List}\n0..*" Tag

Task "0..*\n                        assignedTasks {ordered, List}" -- "0..1" Collaborator   
Project "1" -- "collaborators {ordered, List}\n0..*" Collaborator 

Task "0..*" -- "0..1" Recurrence : has-a >

TaskManager "1" -- "0..*" Task : tasks\n{ordered, List}
TaskManager --> CSVImporter
TaskManager --> CSVExporter
