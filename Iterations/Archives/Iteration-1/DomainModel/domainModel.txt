' =========================
'   ENUMERATIONS
' =========================

enum TaskStatus {
  open
  completed
  cancelled
}

enum PriorityLevel {
  low
  medium
  high
}

' =========================
'   CORE DOMAIN CLASSES
' =========================

class Task {
  id : int
  title : String
  description : String [0..1]
  creationDate : DateTime
  dueDate : DateTime [0..1]
  priority : PriorityLevel
  status : TaskStatus
}

class Project {
  id : int
  name : String
  description : String [0..1]
}

class Subtask {
  title : String
  status : TaskStatus
}

class Tag { 
  id : int
  name : String
}

class ActivityEntry {
  timestamp : DateTime
  description : String
}

' =========================
'   ASSOCIATIONS
' =========================

' Task ↔ Project (optional)
Task "0..*" -- "0..1" Project 

' Task ↔ Subtask (composition)
Task "1" *-- "0..*" Subtask 

' Task ↔ Tag (many-to-many)
Task "0..*" -- "0..*" Tag 

' Task ↔ ActivityEntry (composition)
Task "1" *-- "0..*" ActivityEntry : records >
@enduml
