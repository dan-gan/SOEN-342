SSDs for critical use cases : { createTask(), viewTasks(), updateTask(), completeTask(), cancelTask() }

SSD: createTask()

@startuml
actor User
participant ":__PTMS__" as System
User -> System : createTask(title, description, dueDate, priority)
alt valid input
    System --> User : success()
else invalid input
    System --> User : error("Invalid input")
End
@enduml

SSD: viewTask()

@startuml
actor User
participant ":__PTMS__" as S

User -> S : viewTasks(criteria)

alt tasks found
    S --> User : taskList(tasks)
else no tasks found
    S --> User : emptyList()
end
@enduml


SSD: updateTask()

@startuml
actor User
participant ":__PTMS__" as S

User -> S : updateTask(taskId, updatedData)

alt task exists and data valid
    S --> User : taskUpdated()
else task not found
    S --> User : error("Task not found")
else invalid data
    S --> User : error("Invalid input")
end
@enduml


SSD: completeTask()

@startuml
actor User
participant ":__PTMS__" as S

User -> S : completeTask(taskId)

alt task exists and task is not complete
    S --> User : taskCompleted()
else task not found
    S --> User : error("Task not found")
else task is already complete
    S --> User : error("Task is already complete")
end
@enduml



SSD: cancelTask()

@startuml
actor User
participant ":__PTMS__" as S

User -> S : cancelTask(taskId)

alt task exists and is not already canceled
    S --> User : taskCanceled()
else task not found
    S --> User : error("Task not found")
else task is already canceled
    S --> User : error("Task is already canceled")
end
@enduml
